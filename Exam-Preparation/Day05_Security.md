## 📚 Core Concepts

### 1. Transparent Data Encryption (TDE)

```sql
-- Enable TDE (database-level, encrypts data at rest)
USE master;
CREATE MASTER KEY ENCRYPTION BY PASSWORD = 'StrongPassword123!';
CREATE CERTIFICATE TDE_Cert WITH SUBJECT = 'TDE Certificate';
USE MyDatabase;
CREATE DATABASE ENCRYPTION KEY
WITH ALGORITHM = AES_256
ENCRYPTION BY SERVER CERTIFICATE TDE_Cert;
ALTER DATABASE MyDatabase SET ENCRYPTION ON;

-- Verify
SELECT name, is_encrypted FROM sys.databases WHERE name = 'MyDatabase';
SELECT * FROM sys.dm_database_encryption_keys WHERE database_id = DB_ID();
```

**TDE Key Points:**

- Encrypts data files, log files, backups at rest
- No application changes required
- Certificate must be backed up for restore on another server
- Minimal performance impact (~2-5%)

### 2. Always Encrypted (Client-Side Encryption)

```sql
-- 1. Create Column Master Key (in Windows Cert Store or Azure Key Vault)
-- In SSMS: Right-click Columns Master Keys → New Column Master Key
-- Store: Windows Certificate Store (Current User or Local Machine)
-- Or Azure Key Vault (recommended for production)

-- 2. Create Column Encryption Key
CREATE COLUMN MASTER KEY CMK_Auto1
WITH (KEY_STORE_PROVIDER_NAME = 'MSSQL_CERTIFICATE_STORE',
      KEY_PATH = 'CurrentUser/My/THUMBPRINT_HERE');

CREATE COLUMN ENCRYPTION KEY CEK_Auto1
WITH VALUES (
    COLUMN_MASTER_KEY = CMK_Auto1,
    ALGORITHM = 'RSA_OAEP',
    ENCRYPTED_VALUE = 0x...  -- Generated by tool
);

-- 3. Encrypt columns (deterministic for equality/search, randomized for sensitive)
CREATE TABLE Patients (
    PatientID INT PRIMARY KEY,
    SSN NVARCHAR(11) COLLATE Latin1_General_BIN2
        ENCRYPTED WITH (COLUMN_ENCRYPTION_KEY = CEK_Auto1,
                        ENCRYPTION_TYPE = DETERMINISTIC,
                        ALGORITHM = 'AEAD_AES_256_CBC_HMAC_SHA_256'),
    MedicalHistory NVARCHAR(MAX)
        ENCRYPTED WITH (COLUMN_ENCRYPTION_KEY = CEK_Auto1,
                        ENCRYPTION_TYPE = RANDOMIZED,
                        ALGORITHM = 'AEAD_AES_256_CBC_HMAC_SHA_256'),
    DateOfBirth DATE
        ENCRYPTED WITH (COLUMN_ENCRYPTION_KEY = CEK_Auto1,
                        ENCRYPTION_TYPE = DETERMINISTIC,
                        ALGORITHM = 'AEAD_AES_256_CBC_HMAC_SHA_256')
);

-- 4. Application connection string
-- Add: Column Encryption Setting=Enabled
-- Driver: Microsoft ODBC Driver 17+ / .NET 4.6.1+
```

**Deterministic vs Randomized:**
| Feature | Deterministic | Randomized |
|---------|---------------|------------|
| Equality search | ✅ Supported | ❌ Not supported |
| Joins/Group By | ✅ Supported | ❌ Not supported |
| Pattern matching | ❌ Not supported | ❌ Not supported |
| Security | Lower (patterns visible) | Higher (different ciphertext) |
| Use Case | SSN, IDs, lookup columns | Medical notes, free text |

### 3. Column-Level Encryption (Legacy - Use Always Encrypted Instead)

```sql
-- Symmetric key encryption (deprecated pattern)
CREATE MASTER KEY ENCRYPTION BY PASSWORD = 'Password123!';
CREATE CERTIFICATE Cert_Encrypt WITH SUBJECT = 'Encryption Cert';
CREATE SYMMETRIC KEY SymKey WITH ALGORITHM = AES_256 ENCRYPTION BY CERTIFICATE Cert_Encrypt;

-- Encrypt
OPEN SYMMETRIC KEY SymKey DECRYPTION BY CERTIFICATE Cert_Encrypt;
UPDATE Table SET EncryptedCol = ENCRYPTBYKEY(KEY_GUID('SymKey'), PlainTextCol);
CLOSE SYMMETRIC KEY SymKey;

-- Decrypt
OPEN SYMMETRIC KEY SymKey DECRYPTION BY CERTIFICATE Cert_Encrypt;
SELECT CONVERT(NVARCHAR(MAX), DECRYPTBYKEY(EncryptedCol)) FROM Table;
CLOSE SYMMETRIC KEY SymKey;
```

### 4. Dynamic Data Masking

```sql
-- Add masking to existing columns
ALTER TABLE Customers
ALTER COLUMN Email ADD MASKED WITH (FUNCTION = 'email()');

ALTER TABLE Customers
ALTER COLUMN Phone ADD MASKED WITH (FUNCTION = 'partial(1,"XXX-XXX-",2)');
-- Shows: 1XX-XXX-XX (first 1, last 2)

ALTER TABLE Customers
ALTER COLUMN SSN ADD MASKED WITH (FUNCTION = 'default()');
-- Shows: xxxx (based on data type)

ALTER TABLE Customers
ALTER COLUMN CreditCard ADD MASKED WITH (FUNCTION = 'random(1000, 9999)');
-- Shows: random number in range

-- Grant UNMASK permission to privileged users
GRANT UNMASK TO [AppAdminRole];
GRANT UNMASK TO [AnalyticsUser];

-- Check masking
SELECT c.name, tbl.name AS table_name, c.masking_function
FROM sys.masked_columns c
JOIN sys.tables tbl ON c.object_id = tbl.object_id;
```

**Masking Functions:**
| Function | Example Output | Use Case |
|----------|----------------|----------|
| `default()` | `xxxx`, `0`, `1900-01-01` | Generic by type |
| `email()` | `aXXX@XXXX.com` | Email addresses |
| `partial(prefix, padding, suffix)` | `123-XX-XXXX` | SSN, phones |
| `random(start, end)` | `5432` | Numeric IDs |

### 5. Row-Level Security (RLS)

```sql
-- 1. Create predicate function (inline TVF for performance)
CREATE SCHEMA Security;
GO

CREATE FUNCTION Security.fn_TenantFilter(@TenantID INT)
RETURNS TABLE
WITH SCHEMABINDING
AS
RETURN SELECT 1 AS Result
WHERE @TenantID = CAST(SESSION_CONTEXT(N'TenantID') AS INT);
-- Or for Entra ID: WHERE @TenantID = CAST(SESSION_CONTEXT(N'EntraTenantID') AS INT)

-- 2. Create security policy
CREATE SECURITY POLICY TenantIsolationPolicy
ADD FILTER PREDICATE Security.fn_TenantFilter(TenantID) ON dbo.Orders,
ADD FILTER PREDICATE Security.fn_TenantFilter(TenantID) ON dbo.Customers,
ADD BLOCK PREDICATE Security.fn_TenantFilter(TenantID) ON dbo.Orders AFTER INSERT,
ADD BLOCK PREDICATE Security.fn_TenantFilter(TenantID) ON dbo.Customers AFTER INSERT;

-- 3. Application sets session context
EXEC sp_set_session_context @key = N'TenantID', @value = 123;

-- Query automatically filtered
SELECT * FROM dbo.Orders;  -- Only returns TenantID = 123

-- Disable RLS temporarily (requires ALTER ANY SECURITY POLICY)
ALTER SECURITY POLICY TenantIsolationPolicy WITH (STATE = OFF);
```

**FILTER vs BLOCK Predicates:**
| Predicate | When Applied | Purpose |
|-----------|--------------|---------|
| `FILTER` | SELECT, UPDATE, DELETE | Silently filters rows |
| `BLOCK` | AFTER INSERT, AFTER UPDATE | Explicitly blocks write (error) |

**Performance Tips:**

- Use inline TVF with `SCHEMABINDING`
- Predicate column should be indexed
- Avoid complex logic in predicate function
- `SESSION_CONTEXT` is session-scoped, fast

### 6. Object-Level Permissions

```sql
-- Principle of least privilege
-- Create roles
CREATE ROLE db_datareader_custom;
CREATE ROLE db_datawriter_custom;
CREATE ROLE db_executor;

-- Grant schema-level
GRANT SELECT ON SCHEMA::Sales TO db_datareader_custom;
GRANT INSERT, UPDATE, DELETE ON SCHEMA::Sales TO db_datawriter_custom;
GRANT EXECUTE ON SCHEMA::dbo TO db_executor;

-- Grant specific object
GRANT SELECT ON dbo.CustomerView TO [ReportingUser];
GRANT EXECUTE ON dbo.usp_GetCustomerOrders TO [AppRole];

-- Column-level permissions
GRANT SELECT (CustomerID, CustomerName, Email) ON dbo.Customers TO [SupportRole];
DENY SELECT (SSN, CreditCard) ON dbo.Customers TO [SupportRole];

-- Check permissions
SELECT * FROM fn_my_permissions('dbo.Orders', 'OBJECT');
EXEC sp_helprotect 'dbo.Orders';

-- Never grant to public!
-- REVOKE ALL ON dbo.SensitiveTable FROM public;  -- If accidentally granted
```

### 7. Passwordless Authentication

```sql
-- Azure SQL: Microsoft Entra ID (formerly Azure AD)
-- 1. Create contained user from Entra ID
CREATE USER [alice@contoso.com] FROM EXTERNAL PROVIDER;
ALTER ROLE db_datareader ADD MEMBER [alice@contoso.com];

-- 2. Create group for app access
CREATE USER [AppGroup] FROM EXTERNAL PROVIDER;
ALTER ROLE db_datawriter ADD MEMBER [AppGroup];

-- 3. Managed Identity for Azure services
-- In Azure: Enable System-Assigned Managed Identity on App Service/Function/VM
-- Then in SQL:
CREATE USER [MyAppService] FROM EXTERNAL PROVIDER;
ALTER ROLE db_datareader ADD MEMBER [MyAppService];

-- Connection string (no password!)
-- Server=tcp:myserver.database.windows.net;Authentication=Active Directory Default;
-- Or for Managed Identity: Authentication=Active Directory Managed Identity
```

**Authentication Methods:**
| Method | Use Case | Connection String |
|--------|----------|-------------------|
| `Active Directory Default` | Local dev, Azure CLI logged in | `Authentication=Active Directory Default` |
| `Active Directory Managed Identity` | Azure resources (App Service, Functions, VM) | `Authentication=Active Directory Managed Identity` |
| `Active Directory Service Principal` | CI/CD pipelines, automation | `Authentication=Active Directory Service Principal;User Id=...;Password=...` |
| `Active Directory Interactive` | MFA, interactive login | `Authentication=Active Directory Interactive` |

### 8. Auditing

```sql
-- Server-level audit (captures all databases)
CREATE SERVER AUDIT ServerAudit
TO FILE (FILEPATH = 'C:\Audits', MAXSIZE = 1 GB, MAX_ROLLOVER_FILES = 10)
WITH (QUEUE_DELAY = 1000, ON_FAILURE = CONTINUE);

CREATE SERVER AUDIT SPECIFICATION ServerAuditSpec
FOR SERVER AUDIT ServerAudit
ADD (FAILED_LOGIN_GROUP),
ADD (SUCCESSFUL_LOGIN_GROUP),
ADD (DATABASE_ROLE_MEMBER_CHANGE_GROUP),
ADD (SERVER_ROLE_MEMBER_CHANGE_GROUP);

ALTER SERVER AUDIT ServerAudit WITH (STATE = ON);
ALTER SERVER AUDIT SPECIFICATION ServerAuditSpec WITH (STATE = ON);

-- Database-level audit
CREATE DATABASE AUDIT SPECIFICATION DBAuditSpec
FOR SERVER AUDIT ServerAudit
ADD (SELECT ON SCHEMA::dbo BY public),
ADD (INSERT, UPDATE, DELETE ON SCHEMA::dbo BY public),
ADD (EXECUTE ON SCHEMA::dbo BY public);

ALTER DATABASE AUDIT SPECIFICATION DBAuditSpec WITH (STATE = ON);

-- Query audit logs
SELECT * FROM sys.fn_get_audit_file('C:\Audits\*.sqlaudit', DEFAULT, DEFAULT);

-- Azure SQL: Built-in auditing to Log Analytics / Storage
-- Portal: SQL Database → Auditing → Enable → Log Analytics Workspace
```

### 9. Secure Model Endpoints & API Endpoints

```sql
-- Secure external model endpoint with Managed Identity
-- For sp_invoke_external_rest_endpoint

-- 1. Create credential with Managed Identity
CREATE DATABASE SCOPED CREDENTIAL ModelEndpointCred
WITH IDENTITY = 'Managed Identity';

-- 2. Call external endpoint (uses MI for auth)
DECLARE @response NVARCHAR(MAX);
EXEC sp_invoke_external_rest_endpoint
    @url = 'https://mymodel.endpoint.com/score',
    @method = 'POST',
    @headers = 'Content-Type: application/json',
    @credential = 'ModelEndpointCred',
    @payload = '{"input": "data"}',
    @response = @response OUTPUT;

-- Secure GraphQL/REST/MCP endpoints (Data API Builder)
-- In dab-config.json:
{
  "runtime": {
    "rest": {
      "enabled": true,
      "cors": { "origins": ["https://app.contoso.com"] }
    },
    "graphql": { "enabled": true }
  },
  "security": {
    "authentication": {
      "provider": "EntraID",
      "audiences": ["api://my-app-id"]
    },
    "authorization": {
      "roles": {
        "reader": ["SELECT"],
        "writer": ["INSERT", "UPDATE", "DELETE"]
      }
    }
  }
}

-- MCP Server Security (Fabric/SQL)
-- Use Entra ID for MCP server authentication
-- Restrict MCP tools to read-only unless explicitly needed
```

## 📝 Practice Questions

### Question 1

Always Encrypted deterministic encryption allows which operations?

A) Equality comparison, JOIN, GROUP BY
B) LIKE pattern matching
C) Range queries (>, <)
D) All of the above

**Answer: A** - Deterministic supports equality, joins, group by. No pattern matching or range queries.

### Question 2

Dynamic Data Masking - which function shows first and last character of a string?

A) `default()`
B) `email()`
C) `partial(1, "XXX", 1)`
D) `random(1, 9)`

**Answer: C** - `partial(prefix, padding, suffix)` shows first `prefix` chars, padding, last `suffix` chars.

### Question 3

RLS FILTER predicate on `Orders` table. User queries `SELECT * FROM Orders`. What happens?

A) Error if no session context set
B) Returns all rows (policy ignored)
C) Returns only rows matching predicate (may be empty)
D) Returns rows where TenantID = NULL

**Answer: C** - FILTER predicate silently filters; returns only matching rows (empty if context not set).

### Question 4

Which permission allows a user to view audit logs?

A) `VIEW SERVER STATE`
B) `CONTROL SERVER`
C) `SELECT` on audit file
D) `AUDIT_ADMIN`

**Answer: A** - `VIEW SERVER STATE` needed for `sys.fn_get_audit_file`; `AUDIT_ADMIN` is for managing audit.

### Question 5

Connection string for Azure SQL with System-Assigned Managed Identity:

A) `Authentication=Active Directory Default`
B) `Authentication=Active Directory Managed Identity`
C) `Authentication=Active Directory Service Principal`
D) `Authentication=Active Directory Interactive`

**Answer: B** - Managed Identity uses this auth type; no User Id/Password needed.

## 🎯 Exam Tricks & Gotchas

| Topic                                | Trick/Gotcha                                                                                                               |
| ------------------------------------ | -------------------------------------------------------------------------------------------------------------------------- |
| **TDE**                              | Certificate backup critical for disaster recovery; `BACKUP CERTIFICATE` to file                                            |
| **Always Encrypted**                 | App driver must support it (ODBC 17+, .NET 4.6.1+, JDBC 6.0+); connection string needs `Column Encryption Setting=Enabled` |
| **Deterministic**                    | Requires BIN2 collation for string columns (`Latin1_General_BIN2`)                                                         |
| **DDM**                              | Does NOT protect against: `SELECT INTO`, `BULK INSERT`, `sys.columns` metadata queries; `UNMASK` permission bypasses       |
| **RLS**                              | Predicate function MUST be inline TVF with SCHEMABINDING for performance; `SESSION_CONTEXT` is session-scoped              |
| **Block Predicate**                  | `AFTER INSERT` prevents inserting rows user can't see; `BEFORE UPDATE` prevents changing tenant                            |
| **Permissions**                      | Schema-level grants scale better than object-level; `DENY` overrides `GRANT`                                               |
| **Entra ID Users**                   | Contained users only (no login needed); `FROM EXTERNAL PROVIDER` syntax                                                    |
| **Auditing**                         | `QUEUE_DELAY` = 0 for synchronous (perf impact); 1000ms default async                                                      |
| **sp_invoke_external_rest_endpoint** | Use `DATABASE SCOPED CREDENTIAL` with Managed Identity for secure auth                                                     |

## 📋 Condensed Cheat Sheet

```sql
-- TDE
CREATE MASTER KEY ENCRYPTION BY PASSWORD='P@ss';
CREATE CERTIFICATE TDE_Cert SUBJECT='TDE';
CREATE DATABASE ENCRYPTION KEY WITH ALGORITHM=AES_256 ENCRYPTION BY SERVER CERTIFICATE TDE_Cert;
ALTER DATABASE DB SET ENCRYPTION ON;

-- ALWAYS ENCRYPTED
CREATE COLUMN MASTER KEY CMK WITH (KEY_STORE_PROVIDER_NAME='MSSQL_CERTIFICATE_STORE', KEY_PATH='CurrentUser/My/...');
CREATE COLUMN ENCRYPTION KEY CEK WITH VALUES (COLUMN_MASTER_KEY=CMK, ALGORITHM='RSA_OAEP', ENCRYPTED_VALUE=0x...);
CREATE TABLE T (Col NVARCHAR(50) COLLATE Latin1_General_BIN2 ENCRYPTED WITH (COLUMN_ENCRYPTION_KEY=CEK, ENCRYPTION_TYPE=DETERMINISTIC, ALGORITHM='AEAD_AES_256_CBC_HMAC_SHA_256'));

-- DYNAMIC DATA MASKING
ALTER TABLE T ALTER COLUMN Email ADD MASKED WITH (FUNCTION='email()');
ALTER TABLE T ALTER COLUMN Phone ADD MASKED WITH (FUNCTION='partial(1,"XXX-XXX-",2)');
GRANT UNMASK TO [Role];

-- RLS
CREATE FUNCTION Sec.fn_Filter(@TenantID INT) RETURNS TABLE WITH SCHEMABINDING AS RETURN SELECT 1 WHERE @TenantID = CAST(SESSION_CONTEXT(N'TenantID') AS INT);
CREATE SECURITY POLICY Pol ADD FILTER PREDICATE Sec.fn_Filter(TenantID) ON dbo.Orders, ADD BLOCK PREDICATE Sec.fn_Filter(TenantID) ON dbo.Orders AFTER INSERT;
EXEC sp_set_session_context N'TenantID', 123;

-- PERMISSIONS
GRANT SELECT ON SCHEMA::Sales TO Role;
GRANT SELECT (Col1, Col2) ON Tbl TO Role;
DENY SELECT (SSN) ON Tbl TO Role;

-- ENTRA ID USER
CREATE USER [user@domain.com] FROM EXTERNAL PROVIDER;

-- AUDITING
CREATE SERVER AUDIT Aud TO FILE (FILEPATH='C:\Audit') WITH (QUEUE_DELAY=1000);
CREATE SERVER AUDIT SPECIFICATION Spec FOR SERVER AUDIT Aud ADD (FAILED_LOGIN_GROUP), ADD (SUCCESSFUL_LOGIN_GROUP);
ALTER SERVER AUDIT Aud WITH (STATE=ON);

-- MANAGED IDENTITY FOR EXTERNAL CALL
CREATE DATABASE SCOPED CREDENTIAL Cred WITH IDENTITY = 'Managed Identity';
EXEC sp_invoke_external_rest_endpoint @url='...', @credential='Cred', @response=@r OUT;
```
